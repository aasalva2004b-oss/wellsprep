                                                          )}
                                                      </div>
                                                      
                                                      {session.notes && (
                                                          <div className="text-sm text-slate-600 bg-amber-50/50 p-3 rounded-xl border border-amber-100/50 italic flex gap-2">
                                                              <Key size={14} className="text-amber-400 shrink-0 mt-0.5" />
                                                              <p>{session.notes}</p>
                                                          </div>
                                                      )}
                                                      
                                                      <div className="flex justify-end mt-1">
                                                          {sessionToDelete === session.id ? (
                                                              <div className="flex items-center gap-2">
                                                                 <span className="text-xs font-bold text-slate-500 mr-2">¿Seguro?</span>
                                                                 <button type="button" onClick={(e) => { e.stopPropagation(); setSessionToDelete(null); }} className="text-xs font-bold text-slate-600 bg-slate-100 hover:bg-slate-200 px-3 py-1.5 rounded-lg">Cancelar</button>
                                                                 <button type="button" onClick={(e) => removeRepaso(session.id, e)} className="text-xs font-bold text-white bg-rose-500 hover:bg-rose-600 px-3 py-1.5 rounded-lg shadow-sm">Confirmar</button>
                                                              </div>
                                                          ) : (
                                                              <button type="button" 
                                                                  onClick={(e) => { e.stopPropagation(); setSessionToDelete(session.id); }}
                                                                  className="flex items-center gap-1.5 text-xs font-bold text-rose-500 hover:text-rose-700 hover:bg-rose-50 px-3 py-2 rounded-lg transition-colors"
                                                              >
                                                                  <Trash2 size={14} /> Eliminar Registro
                                                              </button>
                                                          )}
                                                      </div>
                                                  </div>
                                                </div>
                                            </div>
                                        </motion.div>
                                    )})}
                                  </AnimatePresence>
                                </div>
                            )}
                        </div>
                    </div>
                 </div>
              )}

           </div>
           
           {/* Global Footer Fixed at Bottom */}
           <div className="p-4 sm:p-6 bg-white border-t border-slate-100 flex justify-end gap-3 shrink-0 z-20">
              <button type="button" 
                onClick={onClose}
                className="px-6 py-3 rounded-xl text-slate-600 font-bold hover:bg-slate-100 transition-colors w-full sm:w-auto"
              >
                Cancelar
              </button>
              <button type="button" 
                onClick={handleSave}
                className="px-8 py-3 rounded-xl bg-slate-900 text-white font-bold hover:bg-black transition-transform active:scale-95 flex items-center justify-center gap-2 shadow-md hover:shadow-xl w-full sm:w-auto"
              >
                <Save size={18} /> Guardar Cambios
              </button>
           </div>
        </div>

      </div>
    </div>
  );
}
